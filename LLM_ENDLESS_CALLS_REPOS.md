# Pattern: Parallel agent calls
async def deliberate(question: str, agents: List[Agent]):
    # Round 1: All agents respond in parallel
    responses = await asyncio.gather(*[
        agent.respond(question) for agent in agents
    ])
    
    # Rounds 2-N: Agents debate each other's responses
    for round in range(MAX_ROUNDS):
        responses = await asyncio.gather(*[
            agent.respond(question, context=responses) 
            for agent in agents
        ])
        
        if consensus_reached(responses):
            break
    
    return synthesize(responses)

# Call volume: N_agents × N_rounds × 2 (initial + context)

### For Research Automation (AutoResearch, AutoResearchClaw)
```python
# Pattern: Research loop
async def research_cycle(topic: str):
    # 1. Literature review (10-50 calls)
    papers = await search_papers(topic)
    summaries = await asyncio.gather(*[
        llm.summarize(paper) for paper in papers
    ])
    
    # 2. Gap analysis (5-10 calls)
    gaps = await llm.analyze_gaps(summaries)
    
    # 3. Hypothesis generation (10-20 calls)
    hypotheses = await asyncio.gather(*[
        llm.generate_hypothesis(gap) for gap in gaps
    ])
    
    # 4. Experiment design (10-30 calls per hypothesis)
    experiments = await asyncio.gather(*[
        llm.design_experiment(hyp) for hyp in hypotheses
    ])
    
    # 5. Paper drafting (20-50 calls)
    paper = await llm.draft_paper(
        summaries, gaps, hypotheses, experiments
    )
    
    return paper

# Total: 55-190 calls per research cycle
```

# Pattern: Streaming conversation
async def conversation_loop(user_input_stream):
    context = []
    
    async for user_input in user_input_stream:
        # Each user input triggers 1-3 LLM calls
        # 1. Understanding (optional)
        intent = await llm.understand(user_input)
        
        # 2. Generation (required)
        response = await llm.generate(user_input, context)
        
        # 3. Refinement (optional)
        if needs_refinement(response):
            response = await llm.refine(response, user_input)
        
        context.append((user_input, response))
        yield response

# Call volume: 1-3 calls per turn × 20-100 turns per hour
# = 20-300 calls per hour of conversation

### For Long-Running Workflows (OpenWork, Agent SOP)
```python
# Pattern: Workflow orchestration
async def execute_workflow(workflow: Workflow):
    state = WorkflowState()
    
    for step in workflow.steps:
        # 1. Plan step (2-5 calls)
        plan = await llm.plan_step(step, state)
        
        # 2. Execute substeps (5-20 calls)
        results = []
        for substep in plan.substeps:
            result = await llm.execute(substep, state)
            results.append(result)
        
        # 3. Validate (1-3 calls)
        validation = await llm.validate(results, step.criteria)
        
        # 4. Adjust if needed (0-10 calls)
        if not validation.passed:
            adjustment = await llm.adjust(results, validation)
            results = await llm.re_execute(adjustment, state)
        
        state.update(results)
    
    return state

# Total: 8-38 calls per step × 5-20 steps = 40-760 calls per workflow
```
